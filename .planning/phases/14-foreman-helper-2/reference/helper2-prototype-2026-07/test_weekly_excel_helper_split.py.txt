def _row(**overrides):
    row = {
        'Work Request #': '90001',
        'Weekly Reference Logged Date': '2026-08-02',
        'Snapshot Date': '2026-08-02',
        'Units Completed?': True,
        'Units Total Price': 125.0,
        'Quantity': 1,
        'CU': 'TEST-CU',
        'Foreman': 'Primary Foreman',
        'Dept #': '100',
        'Job #': 'JOB-P',
        '__effective_user': 'Primary Foreman',
        '__assignment_method': 'FOREMAN_COLUMN',
        '__source_sheet_id': 123,
        '__row_id': 456,
        '__is_helper_row': False,
    }
    row.update(overrides)
    return row

def _variants(groups):
    return [rows[0]['__variant'] for rows in groups.values()]


def test_no_helper_stays_primary():
    groups = group_source_rows([_row()])
    assert _variants(groups) == ['primary']


def test_helper1_only_is_unchanged_when_helper2_is_enabled(monkeypatch):
    row = _row(
        __is_helper_row=True,
        __helper_foreman='Helper One',
        __helper_dept='200',
        __helper_job='JOB-H1',
    )
    monkeypatch.setattr(gwp, 'HELPER2_ENABLED', False)
    legacy = copy.deepcopy(dict(group_source_rows([copy.deepcopy(row)])))
    monkeypatch.setattr(gwp, 'HELPER2_ENABLED', True)
    enabled = copy.deepcopy(dict(group_source_rows([copy.deepcopy(row)])))
    assert enabled == legacy
    assert _variants(enabled) == ['helper']


def test_helper2_only_emits_a_separate_role():
    row = _row(
        __is_helper2_row=True,
        __helper2_foreman='Helper Two',
        __helper2_email='helper2@example.com',
        __helper2_dept='300',
        __helper2_job='JOB-H2',
    )
    groups = group_source_rows([row])
    assert _variants(groups) == ['helper2']
    output_row = next(iter(groups.values()))[0]
    assert output_row['__helper_foreman'] == 'Helper Two'
    assert output_row['__helper_email'] == 'helper2@example.com'
    assert output_row['__helper_dept'] == '300'
    assert output_row['__helper_job'] == 'JOB-H2'
    assert output_row['__helper_role'] == 2


@pytest.mark.parametrize(
    ('helper1_completed', 'helper2_completed', 'expected'),
    [
        (True, False, 'helper'),
        (False, True, 'helper2'),
    ],
)
def test_completion_checkboxes_are_independent(
    helper1_completed,
    helper2_completed,
    expected,
):
    groups = group_source_rows([
        _row(
            __is_helper_row=helper1_completed,
            __helper_foreman='Helper One',
            __helper_dept='200',
            __helper_job='JOB-H1',
            __is_helper2_row=helper2_completed,
            __helper2_foreman='Helper Two',
            __helper2_email='helper2@example.com',
            __helper2_dept='300',
            __helper2_job='JOB-H2',
        )
    ])
    assert _variants(groups) == [expected]
    assert sum(len(rows) for rows in groups.values()) == 1


def test_dual_completion_fails_closed():
    row = _row(
        __is_helper_row=True,
        __helper_foreman='Helper One',
        __helper_dept='200',
        __is_helper2_row=True,
        __helper2_foreman='Helper Two',
        __helper2_dept='300',
        __helper_completion_conflict=True,
    )
    with pytest.raises(HelperAssignmentConflictError):
        group_source_rows([row])

def test_helper2_filename_identity_round_trips():
    assert build_group_identity(
        'WR_90001_WeekEnding_080226_120000_Helper2_Helper_Two_abc.xlsx'
    ) == ('90001', '080226', 'helper2', 'Helper_Two')
    assert build_group_identity(
        'WR_90001_WeekEnding_080226_120000_'
        'AEPBillable_Helper2_Helper_Two_abc.xlsx'
    ) == ('90001', '080226', 'aep_billable_helper2', 'Helper_Two')
    assert build_group_identity(
        'WR_90001_WeekEnding_080226_120000_'
        'AEPBillable_Helper_Pat_Helper2_abc.xlsx'
    ) == ('90001', '080226', 'aep_billable_helper', 'Pat_Helper2')


def test_helper2_workbook_uses_role2_foreman_department_and_job(
    monkeypatch,
    tmp_path,
):
    monkeypatch.setattr(gwp, 'OUTPUT_FOLDER', str(tmp_path))
    row = _row(
        **{
            'Customer Name': 'AEP',
            '__variant': 'helper2',
            '__current_foreman': 'Helper Two',
            '__helper_foreman': 'Helper Two',
            '__helper_email': 'helper2@example.com',
            '__helper_dept': '300',
            '__helper_job': 'JOB-H2',
            '__week_ending_date': dt.datetime(2026, 8, 2),
        }
    )
    result = gwp.generate_excel(
        '080226_90001_HELPER2_Helper_Two',
        [row],
        dt.datetime(2026, 8, 2),
        data_hash='abcdef0123456789',
    )
    path, filename = result[0], result[1]
    assert os.path.exists(path)
    assert '_Helper2_Helper_Two_' in filename

    workbook = openpyxl.load_workbook(path, read_only=True)
    worksheet = workbook.active
    details = {
        worksheet.cell(row=index, column=6).value:
        worksheet.cell(row=index, column=7).value
        for index in range(1, worksheet.max_row + 1)
    }
    assert details['Foreman:'] == 'Helper Two'
    assert details['Dept #:'] == '300'
    assert details['Job #:'] == 'JOB-H2'


def test_helper2_email_and_job_participate_in_role2_hash_only():
    first = _row(
        __variant='helper2',
        __current_foreman='Helper Two',
        __helper_foreman='Helper Two',
        __helper_email='first@example.com',
        __helper_dept='300',
        __helper_job='JOB-H2',
    )
    second = copy.deepcopy(first)
    second['__helper_email'] = 'second@example.com'
    assert calculate_data_hash([first]) != calculate_data_hash([second])

    first['__variant'] = 'helper'
    second['__variant'] = 'helper'
    assert calculate_data_hash([first]) == calculate_data_hash([second])
