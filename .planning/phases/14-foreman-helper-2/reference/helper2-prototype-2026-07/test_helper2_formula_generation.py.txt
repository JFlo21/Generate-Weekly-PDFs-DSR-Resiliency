def _column(
    column_id,
    title,
    column_type='TEXT_NUMBER',
    formula=None,
):
    return {
        'id': column_id,
        'index': column_id,
        'title': title,
        'type': column_type,
        'options': [],
        'contact_options': [],
        'hidden': False,
        'locked': False,
        'formula': formula,
    }

def test_formula_clone_changes_only_local_and_reference_names():
    original = (
        '=IF([Helping Foreman Completed Unit?]@row = 1, '
        'INDEX({Roster Dept}, MATCH([Foreman Helping?]@row, '
        '{Roster Name}, 0)))'
    )
    cloned = clone_helper1_formula(
        original,
        {
            'Roster Dept': 'H2 Roster Dept 000500',
            'Roster Name': 'H2 Roster Name 000500',
        },
    )
    assert original == (
        '=IF([Helping Foreman Completed Unit?]@row = 1, '
        'INDEX({Roster Dept}, MATCH([Foreman Helping?]@row, '
        '{Roster Name}, 0)))'
    )
    assert '[Helping Foreman #2 Completed Unit?]@row' in cloned
    assert '[Foreman Helping? #2]@row' in cloned
    assert '{H2 Roster Dept 000500}' in cloned


def test_formula_clone_rejects_at_cell():

def test_automation_inventory_excludes_recipient_details():
    rule = SimpleNamespace(
        id=77,
        name='Helper assignment metadata',
        enabled=True,
        user_can_modify=True,
        disabled_reason=None,
        action=SimpleNamespace(
            type='NOTIFICATION_ACTION',
            recipients=['private@example.com'],
        ),
        created_by=SimpleNamespace(email='private@example.com'),
    )
    sheets = SimpleNamespace(
        list_automation_rules=lambda sheet_id, include_all: (
            SimpleNamespace(data=[rule])
        )
    )
    client = SmartsheetClient(SimpleNamespace(Sheets=sheets))
    result = client.list_automation_rules(3733355007790980)
    assert result == [{
        'id': 77,
        'name': 'Helper assignment metadata',
        'enabled': True,
        'user_can_modify': True,
        'disabled_reason': '',
        'action_type': 'NOTIFICATION_ACTION',
    }]


def test_plan_appends_all_columns_and_uses_distinct_cross_references():
    plan = plan_migration(_inventory())
    sheet = plan['sheet_plans'][0]
    assert sheet['status'] == 'planned'
    assert [action['title'] for action in sheet['actions']] == list(
        HELPER2_TITLES
    )
    assert all(
        ref['name'].startswith('H2 ')
        for ref in sheet['cross_sheet_reference_writes']
    )
    assigned = next(
        ref
        for ref in sheet['cross_sheet_reference_writes']
        if ref['name'].startswith('H2 Assigned Helper 2')
    )
    assert assigned['start_column_id'] == 103
    foreman = next(
        ref
        for ref in sheet['cross_sheet_reference_writes']
        if ref['name'].startswith('H2 RA Foreman Helping')
    )
    assert foreman['start_column_id'] == 104
    email_action = next(
        action
        for action in sheet['actions']
        if action['title'] == 'Foreman Helper #2 Email'
    )
    assert '{' + assigned['name'] + '}' in email_action['formula']
    assert '{Assigned Helper 1}' not in email_action['formula']
    assert not plan['blockers']

def test_duplicate_assignment_aliases_fail_closed():
    inventory = _inventory()
    resource = inventory['resource_analyst']
    resource['columns'].append(
        _column(106, 'Assigned Helper? 2', 'MULTI_CONTACT_LIST')
    )
    resource['assignment_contract']['assignment_aliases_present'] = [
        'Assigned Helper 2?',
        'Assigned Helper? 2',
    ]
    plan = plan_migration(inventory)
    assert 'AMBIGUOUS_RESOURCE_HELPER2_ASSIGNMENT_COLUMN' in plan['blockers']


def test_existing_helper2_columns_are_never_renamed_or_overwritten():
    columns = _helper1_columns()
    columns.extend(
        _column(1000 + index, title)
        for index, title in enumerate(HELPER2_TITLES[1:], start=1)
    )
    plan = plan_migration(_inventory(columns=columns))
    actions = plan['sheet_plans'][0]['actions']
    append_actions = [
        action for action in actions if action['action'] == 'append_column'
    ]
    assert [action['title'] for action in append_actions] == [
        'Foreman Helper #2 Email'
    ]
    assert all(action['action'] != 'rename_column' for action in actions)

def test_legacy_and_current_schema_classification():
    assert classify_schema([_column(1, 'Work Request #')]) == (
        'legacy_or_unknown'
    )
    current = _helper1_columns() + [
        _column(1000 + index, title)
        for index, title in enumerate(HELPER2_TITLES)
    ]
    assert classify_schema(current) == 'helper2_complete'

def test_write_requires_all_independent_gates(monkeypatch):
    plan = plan_migration(_inventory())
    monkeypatch.delenv('PRODUCTION_SMARTSHEET_WRITE_APPROVED', raising=False)
    with pytest.raises(PermissionError):
        assert_execution_approved(
            plan,
            execute=True,
            tests_passed=True,
            approved_plan_sha256=plan['plan_sha256'],
        )


def test_formula_error_values_are_reported_by_validator():
    inventory = _inventory()
    columns = copy.deepcopy(inventory['sheets'][0]['columns'])
    columns.extend(
        _column(1000 + index, title)
        for index, title in enumerate(HELPER2_TITLES)
    )
    columns[-1]['formula'] = '=#REF'
    inventory['sheets'][0]['columns'] = columns
    result = validate_inventory(inventory)
    assert result['summary']['invalid_sheets'] == 1
    assert result['sheet_results'][0]['formula_error_columns'] == [
        'Helper #2 Job [#]'
    ]


def test_cli_dry_run_writes_reviewable_plan_without_a_client(
    monkeypatch,
    tmp_path,
):
    inventory_path = tmp_path / 'inventory.json'
    plan_path = tmp_path / 'plan.json'
    inventory_path.write_text(
        json.dumps(_inventory()),
        encoding='utf-8',
    )
    monkeypatch.setattr(
        sys,
        'argv',
        [
            'helper2_migrate',
            '--inventory',
            str(inventory_path),
            '--plan-output',
            str(plan_path),
        ],
    )
    assert migration_main() == 0
    written = json.loads(plan_path.read_text(encoding='utf-8'))
    assert written['mode'] == 'dry-run'
    assert written['plan_sha256']
